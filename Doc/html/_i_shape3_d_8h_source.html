122  template <class T> const bool IShape3D<T>::IsCollision(const IShape3D * shape) const {
123  // Get the plane perpendicular to the shapes position in order to get the inner depth
124 
125  // Get the z axis vector for the plane
126  CVector3F zAxis = shape->GetPosition() - this->m_position;
127  CVector3F tmpYAxis = CVector3F(0, 1, 0);
128 
129  // Get the x axis vector for the plane
130  CVector3F xAxis = tmpYAxis.CrossProduct(zAxis);
131 
132  // Get the y axis vector for the plane
133  CVector3F yAxis = zAxis.CrossProduct(xAxis);
134 
135  CVector3F vertex[] = { (this->m_position + xAxis), this->m_position, (this->m_position + yAxis) };
136  CPlane3DF plane(vertex);
137 
138  // Get the distance between the two centers
139  float distance = this->m_position.Distance(shape->GetPosition());
140 
141  return distance <= (GetInnerDepth(plane) + shape->GetInnerDepth(plane));
142  }
