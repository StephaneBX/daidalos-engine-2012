00182         template <class T> const float CAABoundingBox<T>::GetInnerDepth(const CPlane3D<T> & plane) const
00183         {
00184                 CVector3<T> endSegment = this->m_position - (plane.GetNormal() * this->m_size.x * this->m_size.y * this->m_size.z);
00185                 float factor = 1.0f;
00186                 CVector3<T> intersection;
00187 
00188                 // Check the intersection between each plane and the segment (plane - position)
00189                 for(auto it = this->m_planes.begin() ; it != this->m_planes.end() ; ++it)
00190                 {
00191                         intersection = (*it).IntersectSegment(this->m_position, endSegment, factor);
00192                         if(factor < 1.0f)
00193                                 break;
00194                 }
00195 
00196                 return (intersection - this->m_position).Length();
00197         }
