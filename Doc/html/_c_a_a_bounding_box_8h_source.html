173  template <class T> const float CAABoundingBox<T>::GetInnerDepth(const CPlane3D<T> & plane) const {
174  CVector3<T> endSegment = this->m_position + CVector3<T>(plane.GetNormal().x * this->m_size.x, plane.GetNormal().y * this->m_size.y, plane.GetNormal().z * this->m_size.z);
175  float factor = 1.0f;
176  CVector3<T> intersection;
177 
178  // Check the intersection between each plane and the segment (plane - position)
179  for(auto it = this->m_planes.begin(); it != this->m_planes.end(); ++it) {
180  intersection = (*it).IntersectSegment(this->m_position, endSegment, factor);
181  if(factor > 0 && factor < 1.0f) {
182  break;
183  }
184  }
185 
186  return (intersection - this->m_position).Length();
187  }
