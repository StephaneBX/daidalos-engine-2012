110  template <class T> CSphere<T> CSphere<T>::GetUnion(const CSphere<T> & sphere) const {
111  // Get the minimum and maximum values
112  CVector3<T> direction = (sphere.GetPosition() - this->m_position).Normalize();
113  CVector3<T> start = this->m_position - (direction * this->m_radius);
114  CVector3<T> end = sphere.GetPosition() + (direction * sphere.GetInnerDepth());
115 
116  return CSphere<T>((start + end) / 2.0f, (end - start).Length() / 2.0f);
117  }
