252  template <class T> inline ERectangleIntersection CRectangle<T>::Intersect(const CVector2<T> & point) const {
253  if(point.x >= this->m_topLeft.x && point.x <= this->m_bottomRight.x && point.y >= this->m_topLeft.y && point.y <= this->m_bottomRight.y)
254  return RECTANGLE_INTERSECTION_INSIDE;
255  else
256  return RECTANGLE_INTERSECTION_OUTSIDE;
257  }

264  template <class T> inline ERectangleIntersection CRectangle<T>::Intersect(const CRectangle<T> & rect) const {
265  CVector2<T> start = CVector2<T>(std::max(this->m_topLeft.x, rect.m_topLeft.x), std::max(this->m_topLeft.y, rect.m_topLeft.y));
266  CVector2<T> end = CVector2<T>(std::min(this->m_bottomRight.x, rect.m_bottomRight.x), std::min(this->m_bottomRight.y, rect.m_bottomRight.y));
267  CRectangle<T> tmp = CRectangle<T>(start, end - start);
268 
269  if(start.x > end.x || start.y > end.y)
270  return RECTANGLE_INTERSECTION_OUTSIDE;
271  else
272  if(tmp == *this || tmp == rect)
273  return RECTANGLE_INTERSECTION_INSIDE;
274  else
275  return RECTANGLE_INTERSECTION_INTERSECT;
276  }
