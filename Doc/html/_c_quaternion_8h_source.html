192  inline const CQuaternion CQuaternion::operator*(const CQuaternion & quat) const {
193  CQuaternion tmp;
194 
195  tmp.x = (quat.w * this->x) + (quat.x * this->w) + (quat.y * this->z) - (quat.z * this->y);
196  tmp.y = (quat.w * this->y) + (quat.y * this->w) + (quat.z * this->x) - (quat.x * this->z);
197  tmp.z = (quat.w * this->z) + (quat.z * this->w) + (quat.x * this->y) - (quat.y * this->x);
198  tmp.w = (quat.w * this->w) - (quat.x * this->x) - (quat.y * this->y) - (quat.z * this->z);
199 
200  return tmp;
201  }

253  inline const CQuaternion CQuaternion::Log() const {
254  CQuaternion tmp;
255  const float theta = acos(this->w);
256  const float sinTheta = sin(theta);
257 
258  if(fabs(sinTheta) > EPSILON) {
259  tmp = (*this * theta) / sinTheta;
260  tmp.w = 0;
261  } else
262  tmp = CQuaternion(0, 0, 0, 0);
263 
264  return tmp;
265  }

270  inline const CQuaternion CQuaternion::Exp() const {
271  CQuaternion tmp;
272  const float theta = sqrt(this->x * this->x + this->y * this->y + this->z * this->z);
273  const float sinTheta = sin(theta);
274  const float cosTheta = cos(theta);
275 
276  tmp.w = cosTheta;
277  if(theta > EPSILON) {
278  tmp.x = this->x * sinTheta / theta;
279  tmp.y = this->y * sinTheta / theta;
280  tmp.z = this->z * sinTheta / theta;
281  }
282 
283  return tmp;
284  }

326  inline const CQuaternion CQuaternion::Slerp(const CQuaternion & quat2, const float t) const {
