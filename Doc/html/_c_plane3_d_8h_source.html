166  template <class T> void CPlane3D<T>::Build(const CVector3<T> * vertex) {
167  // Normal calculation
168  CVector3F ab = vertex[1] - vertex[0];
169  CVector3F ac = vertex[2] - vertex[0];
170  this->m_normal = ab.CrossProduct(ac).Normalize();
171 
172  // Constant calculation
173  this->m_constant = -(vertex[1].x * this->m_normal.x + vertex[1].y * this->m_normal.y + vertex[1].z * this->m_normal.z);
174  }

176  template <class T> const float CPlane3D<T>::PointDistance(const CVector3<T> & point) const {
177  return this->m_constant + this->m_normal.DotProduct(point);
178  }

180  template <class T> EPointOrientation CPlane3D<T>::ClassifyPoint(const CVector3<T> & point) const {
181  float result = PointDistance(point);
182  EPointOrientation orientation = POINT_ORIENTATION_ON_PLANE;
183  if(result > EPSILON) {
184  orientation = POINT_ORIENTATION_POSITIVE_SIDE;
185  } else {
186  if(result < -EPSILON)
187  orientation = POINT_ORIENTATION_NEGATIVE_SIDE;
188  }
189 
190  return orientation;
191  }

193  template <class T> CVector3<T> CPlane3D<T>::IntersectSegment(const CVector3<T> & p1, const CVector3<T> & p2, float & k) const {
194  CVector3F u, result;
195  u = p1 - p2;
196 
197  k = (this->m_normal.x * p1.x + this->m_normal.y * p1.y + this->m_normal.z * p1.z + this->m_constant) / (this->m_normal.x * u.x + this->m_normal.y * u.y + this->m_normal.z * u.z);
198 
199  result.x = p1.x - k * u.x;
200  result.y = p1.y - k * u.y;
201  result.z = p1.z - k * u.z;
202 
203  return result;
204  }
