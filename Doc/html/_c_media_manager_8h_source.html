106  template <class T> void CMediaManager::AddLoader(ILoader<T> * loader, const std::string & extensions) {
107  CTokenizer tokenizer;
108  tokenizer.AddLine(extensions);
109  tokenizer.Tokenize(" ");
110 
111  std::set<ILoader<T>*> replacedLoaders;
112  for(unsigned int i = 0; i < tokenizer.GetTokens().size(); ++i) {
113 
114  // Store the loaders that are replaced
115  if(CLoaderExtensions<T>::s_loaders.find(tokenizer.GetTokens()[i]) != CLoaderExtensions<T>::s_loaders.end()) {
116  replacedLoaders.insert(CLoaderExtensions<T>::s_loaders[tokenizer.GetTokens()[i]]);
117  }
118 
119  // Set or replace
120  CLoaderExtensions<T>::s_loaders[tokenizer.GetTokens()[i]] = loader;
121  }
122 
123  // If the replaced loaders are not used anymore, delete them
124  for(auto it = replacedLoaders.begin(); it != replacedLoaders.end(); it++) {
125  bool found = false;
126  for(auto it2 = CLoaderExtensions<T>::s_loaders.begin(); it2 != CLoaderExtensions<T>::s_loaders.end() && !found; it2++) {
127  if((*it) == (*it2).second) {
128  found = true;
129  }
130  }
131 
132  if(!found) {
133  delete (*it);
134  }
135  }
136  }

138  template <class T> void CMediaManager::RemoveLoader() {
139 
140  // Delete all the loaders
141  std::set<ILoader<T>*> deletedLoaders;
142  for(auto it = CLoaderExtensions<T>::s_loaders.begin(); it != CLoaderExtensions<T>::s_loaders.end(); it++) {
143  if(deletedLoaders.find((*it).second) == deletedLoaders.end()) {
144  deletedLoaders.insert((*it).second);
145  delete (*it).second;
146  }
147  }
148 
149  CLoaderExtensions<T>::s_loaders.clear();
150  }
