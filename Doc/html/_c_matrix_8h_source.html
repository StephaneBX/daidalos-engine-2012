126  inline CMatrix::CMatrix(const CVector3F & scale, const CVector3F & rotation, const CVector3F & translation) {
127  CMatrix scaleMatrix;
128  CMatrix rotationMatrix;
129  CMatrix translationMatrix;
130  CQuaternion rot;
131 
132  scaleMatrix = scaleMatrix.SetScale(scale.x, scale.y, scale.z);
133  CQuaternion::YawPitchRoll(rot, rotation.x, rotation.y, rotation.z);
134  rotationMatrix = RotationQuaternion(rot);
135  translationMatrix = translationMatrix.SetTranslation(translation.x, translation.y, translation.z);
136 
137  *this = scaleMatrix * rotationMatrix * translationMatrix;
138  }

236  inline const CMatrix CMatrix::Inverse() const {
237  CMatrix ret;
238 
239  float a0 = this->_11 * this->_22 - this->_12 * this->_21;
240  float a1 = this->_11 * this->_23 - this->_13 * this->_21;
241  float a2 = this->_11 * this->_24 - this->_14 * this->_21;
242  float a3 = this->_12 * this->_23 - this->_13 * this->_22;
243  float a4 = this->_12 * this->_24 - this->_14 * this->_22;
244  float a5 = this->_13 * this->_24 - this->_14 * this->_23;
245  float b0 = this->_31 * this->_42 - this->_32 * this->_41;
246  float b1 = this->_31 * this->_43 - this->_33 * this->_41;
247  float b2 = this->_31 * this->_44 - this->_34 * this->_41;
248  float b3 = this->_32 * this->_43 - this->_33 * this->_42;
249  float b4 = this->_32 * this->_44 - this->_34 * this->_42;
250  float b5 = this->_33 * this->_44 - this->_34 * this->_43;
251 
252  float determinant = a0 *b5 - a1 * b4 + a2 * b3 + a3 * b2 - a4 * b1 + a5 * b0;
253 
254  if(std::fabs(determinant) > EPSILON) {
255  ret._11 = this->_22 * b5 - this->_23 * b4 + this->_24 * b3;
256  ret._21 = -this->_21 * b5 + this->_23 * b2 - this->_24 * b1;
257  ret._31 = this->_21 * b4 - this->_22 * b2 + this->_11 * b0;
258  ret._41 = -this->_21 * b3 + this->_22 * b1 - this->_23 * b0;
259  ret._12 = -this->_12 * b5 + this->_13 * b4 - this->_14 * b3;
260  ret._22 = this->_11 * b5 - this->_13 * b2 + this->_14 * b1;
261  ret._32 = -this->_11 * b4 + this->_12 * b2 - this->_14 * b0;
262  ret._42 = this->_11 * b3 - this->_12 * b1 + this->_13 * b0;
263  ret._13 = this->_42 * a5 - this->_43 * a4 + this->_44 * a3;
264  ret._23 = -this->_41 * a5 + this->_43 * a2 - this->_44 * a1;
265  ret._33 = this->_41 * a4 - this->_42 * a2 + this->_44 * a0;
266  ret._43 = -this->_41 * a3 + this->_42 * a1 - this->_43 * a0;
267  ret._14 = -this->_32 * a5 + this->_33 * a4 - this->_34 * a3;
268  ret._24 = this->_31 * a5 - this->_33 * a2 + this->_34 * a1;
269  ret._34 = -this->_31 * a4 + this->_32 * a2 - this->_34 * a0;
270  ret._44 = this->_31 * a3 - this->_32 * a1 + this->_33 * a0;
271 
272  float invDet = 1 / determinant;
273  ret *= invDet;
274  }
275 
276  return ret;
277  }

473  inline const CQuaternion CMatrix::TransformRotation() const {
474  CQuaternion tmp;
475  const float trace = this->_11 + this->_22 + this->_33;
476 
477  if(trace > EPSILON) {
478  float scale = 0.5f / sqrtf(trace + 1.0f);
479  tmp.w = 0.25f / scale;
480  tmp.x = (this->_23 - this->_32) * scale;
481  tmp.y = (this->_31 - this->_13) * scale;
482  tmp.z = (this->_12 - this->_21) * scale;
483  } else {
484  if(this->_11 > this->_22 && this->_11 > this->_33) {
485  const float scale = 2.0f * sqrtf(1.0f + this->_11 - this->_22 - this->_33);
486  tmp.w = (this->_23 - this->_32) / scale;
487  tmp.x = 0.25f * scale;
488  tmp.y = (this->_21 + this->_12) / scale;
489  tmp.z = (this->_31 + this->_13) / scale;
490  } else {
491  if(this->_22 > this->_33) {
492  const float scale = 2.0f * sqrtf(1.0f + this->_22 - this->_11 - this->_33);
493  tmp.w = (this->_31 - this->_13) / scale;
494  tmp.x = (this->_21 + this->_12) / scale;
495  tmp.y = 0.25f * scale;
496  tmp.z = (this->_32 + this->_23) / scale;
497  } else {
498  const float scale = 2.0f * sqrtf(1.0f + this->_33 - this->_11 - this->_22);
499  tmp.w = (this->_12 - this->_21) / scale;
500  tmp.x = (this->_31 + this->_13) / scale;
501  tmp.y = (this->_32 + this->_23) / scale;
502  tmp.z = 0.25f * scale;
503  }
504  }
505  }
506 
507  return tmp.Normalize();
508  }
