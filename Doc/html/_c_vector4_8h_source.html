345  template <class T> inline const CVector4<T> CVector4<T>::GetFrom(const std::string text, const CVector4<T> & defaultValue, const std::string separator) {
346  CVector4<T> result;
347  CTokenizer token;
348  token.AddLine(text);
349 
350  if(text.length() > 0 && token.Tokenize(separator, 4) && token.GetTokens().size() == 4) {
351  std::istringstream ss;
352 
353  ss.clear();
354  ss.str(token.GetTokens()[0]);
355  ss >> result.x;
356 
357  ss.clear();
358  ss.str(token.GetTokens()[1]);
359  ss >> result.y;
360 
361  ss.clear();
362  ss.str(token.GetTokens()[2]);
363  ss >> result.z;
364 
365  ss.clear();
366  ss.str(token.GetTokens()[3]);
367  ss >> result.w;
368  } else {
369  result = defaultValue;
370  }
371 
372  return result;
373  }
